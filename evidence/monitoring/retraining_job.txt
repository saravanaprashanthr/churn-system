metadata:
  annotations:
  labels:
    app.kubernetes.io/name: churn-retrain
  name: churn-retrain
  namespace: churn
spec:
  concurrencyPolicy: Forbid
  failedJobsHistoryLimit: 3
  jobTemplate:
    metadata: {}
    spec:
      activeDeadlineSeconds: 3600
      backoffLimit: 0
      template:
        metadata:
          labels:
            app.kubernetes.io/name: churn-retrain
        spec:
          containers:
          - args:
            - |
              set -e
              python -m training.train --optuna-trials "${OPTUNA_TRIALS}" --out-dir /tmp/training
              set +e
              python -m training.promote --candidate models:/churn-90d@candidate --out-dir /tmp/promotion
              code=$?
              echo "promotion gate exit code: $code (0 promoted, 1 rejected, 2 error)"
              [ "$code" -le 1 ]
            command:
            - /bin/sh
            - -c
            env:
            - name: OPTUNA_TRIALS
              value: "20"
            - name: MLFLOW_TRACKING_URI
              value: http://host.docker.internal:5001
            - name: DATABASE_URL
              valueFrom:
                secretKeyRef:
                  key: DATABASE_URL
                  name: churn-retrain
            image: churn-trainer:c6362bf
            imagePullPolicy: IfNotPresent
            name: retrain
            resources:
              limits:
                cpu: "4"
                memory: 2Gi
              requests:
                cpu: "2"
                memory: 1Gi
            securityContext:
              allowPrivilegeEscalation: false
              capabilities:
                drop:
                - ALL
            terminationMessagePath: /dev/termination-log
            terminationMessagePolicy: File
          dnsPolicy: ClusterFirst
          restartPolicy: Never
          schedulerName: default-scheduler
          securityContext:
            runAsNonRoot: true
            runAsUser: 1000
            seccompProfile:
              type: RuntimeDefault
          terminationGracePeriodSeconds: 30
      ttlSecondsAfterFinished: 604800
  schedule: 0 3 * * 1
  successfulJobsHistoryLimit: 3
  suspend: false
